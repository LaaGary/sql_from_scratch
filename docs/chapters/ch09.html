Chapter 09 / JOIN

Connecting two tables: JOIN and nested loops

Real databases rarely have just one table: books have authors, orders have customers, comments have articles. JOIN is the operation that pairs rows from two tables by some condition. Its implementation is surprisingly simple — two loops — but that's also where the performance problems start.

JOIN semantics

books JOIN authors ON books.author_id = authors.id means: for each row in books, find every authors row whose "author_id equals authors.id" and pair it up. If a book matches two authors (which shouldn't happen in theory, but can without a foreign key constraint), you get two rows.

TypeBehaviorOur support
JOIN / INNER JOINkeep only rows that match on both sides✅
LEFT JOINleft rows are always kept; an unmatched right side is filled with NULL✅
RIGHT JOIN / FULL JOINkeep the reverse side, or both sides❌ (try it yourself)

Nested loops: the most obvious implementation

In pseudocode: 

for (each row on the left) {
  let matched = false;
  for (each row on the right) {
    if (ON condition holds) {
      output left + right;
      matched = true;
    }
  }
  if (!matched && is LEFT JOIN) {
    output left + right side all NULL;
  }
}

Our scan() uses the same idea, except the generator yields one row at a time so we never have to materialize every combination in memory.

Why do columns need aliases after a JOIN?

Both tables may have id. If you write SELECT id FROM books JOIN authors ..., the engine can't tell which one you mean, and the correct response is to raise: 

AMBIGUOUS_COLUMN: ambiguous column name: id exists in both books and authors
Hint: add a table alias, for example b.id

This check was actually written in Chapter 6's evaluateColumn(): it collects all matching sources, reports "column not found" for zero and "ambiguous" for more than one. JOIN is just what makes the situation real.

Changes to the query pipeline

The changes are concentrated in two places: buildSources() has to collect the JOIN sources (including the duplicate-alias check), and scan() has to run the nested loop. WHERE, projection, grouping, and sorting don't change at all — they just see "rows with more sources".

   * A SELECT goes through these stages in order:
   *

  function applyWhere(ast, rows) {
    if (!ast.where) return rows;
    const result = [];
    for (const row of rows) {
      // Three-valued logic: keep only TRUE (NULL / UNKNOWN count as not satisfied)
      if (truthy(evaluate(ast.where, row, {}))) result.push(row);
    }
    return result;
  }
  
  // ---------------------------------------------------------------------------
  // Grouping and aggregation
  // ---------------------------------------------------------------------------
  
  /** Collect all aggregate expressions in SELECT / HAVING / ORDER BY (deduplicated). */
  function collectAggregateExpressions(ast) {
    const found = new Map();
    const visit = (node) => {
      if (node && typeof node === "object" && node.type === "call" && AGGREGATE_FUNCTIONS.has(node.name.toUpperCase())) {
        found.set(JSON.stringify(node), node);
      }
    };
    walk(ast.columns.map((item) => item.expr), visit);
    walk(ast.having, visit);
    walk(ast.orderBy.map((order) => order.expr), visit);
    return [...found.values()];
  }
  
  function walk(node, visit) {
    if (!node || typeof node !== "object") return;
    if (Array.isArray(node)) {
      for (const item of node) walk(item, visit);
      return;
    }
    visit(node);
    for (const value of Object.values(node)) {
      if (value && typeof value === "object") walk(value, visit);
    }
  }
  
  /**
   * Group rows according to GROUP BY.
   * With aggregates but no GROUP BY (e.g. SELECT COUNT(*) FROM books),
   * the whole table is one group; even with no rows, emit an empty group so COUNT(*) is 0.
   */
  function buildGroups(ast, rows, aggregateExpressions) {
    const groups = [];
  
    if (ast.groupBy.length === 0) {
      groups.push({ row: rows[0] ?? [], rows });
    } else {
      const index = new Map();
      for (const row of rows) {
        const keyValues = ast.groupBy.map((expr) => evaluate(expr, row, {}));
        const key = groupKey(keyValues);
        let group = index.get(key);
        if (!group) {
          group = { row, rows: [] };
          index.set(key, group);
          groups.push(group);
        }
        group.rows.push(row);
      }
    }
  
    for (const group of groups) {
      group.state = createAggregateState(aggregateExpressions, group.rows);
    }
    return groups;
  }

  // Output columns
  // ---------------------------------------------------------------------------
  
  /**
   * Determine the output columns. SELECT * expands into multiple column definitions;
   * other expressions use their alias or a derived name.
   */
  function buildProjection(ast, sources) {
    const defs = [];
    for (const item of ast.columns) {
      if (item.expr.type === "star") {
        const targets = item.expr.table
          ? sources.filter((source) => source.alias.toLowerCase() === item.expr.table.toLowerCase())
          : sources;
        if (targets.length === 0) {
          throw new SqlError("Table not found: " + (item.expr.table ?? "(no FROM)"), {
            code: "SCHEMA_ERROR",
            hint: "SELECT * requires FROM.",
          });
        }
        for (const source of targets) {
          for (const column of source.table.columns) {
            defs.push({
              kind: "source-column",
              alias: source.alias,
              columnName: column.name,
              label: column.name,
            });
          }
        }
        continue;
      }
      defs.push({
        kind: "expression",
        expr: item.expr,
        label: item.alias ?? columnLabel(item.expr),
      });
    }
    if (defs.length === 0) {
      throw new SqlError("SELECT requires at least one column", { code: "SYNTAX_ERROR" });
    }
    return defs;
  }
  
  function projectUnit(defs, unit) {
    return defs.map((def) => {
      if (def.kind === "source-column") {
        const source = unit.row.find((item) => item.alias === def.alias);
        if (!source || !source.values) return null;
        return source.values[def.columnName] ?? null;
      }
      return evaluate(def.expr, unit.row, unit.state);
    });
  }
  
  /** Output column name and type (the type comes from the first non-NULL value). */
  function describeColumns(defs, projected) {
    return defs.map((def, index) => {
      let type = "null";
      for (const item of projected) {
        if (!isNull(item.values[index])) {
          type = typeOf(item.values[index]);
          break;
        }
      }
      return { name: def.label, type };
    });
  }
  
  function dedupe(projected) {
    const seen = new Set();
    const result = [];
    for (const item of projected) {
      const key = groupKey(item.values);
      if (seen.has(key)) continue;
      seen.add(key);
      result.push(item);
    }
    return result;
  }
  
  // ---------------------------------------------------------------------------
  // Sorting and LIMIT
  // ---------------------------------------------------------------------------
  
  /**
   * Sorting. ORDER BY accepts source columns, output column aliases, or aggregate functions:
   *
   *   SELECT author, COUNT(*) AS n FROM books GROUP BY author ORDER BY n DESC
   */
  function sortRows(projected, orderBy) {
    const decorated = projected.map((item) => ({
      ...item,
      keys: orderBy.map((order) => orderValue(order.expr, item)),
    }));
  
    decorated.sort((left, right) => {
      for (let i = 0; i < orderBy.length; i++) {
        const direction = orderBy[i].direction === "desc" ? -1 : 1;
        const compared = compareValues(left.keys[i], right.keys[i]);
        if (compared !== 0) return compared * direction;
      }
      return 0;
    });
  
    return decorated;
  }
  
  function orderValue(expr, item) {
    // Check for an output column alias first (SQL allows aliases in ORDER BY).
    if (expr.type === "column" && !expr.table) {
      const index = item.labels.findIndex((label) => label.toLowerCase() === expr.name.toLowerCase());
      if (index >= 0) return item.values[index];
    }
    return evaluate(expr, item.row, item.unit.state);
  }
  
  function applyLimit(projected, ast) {
    const offset = ast.offset ?? 0;
    const limit = ast.limit;
    if (limit === null && offset === 0) return projected;
    const end = limit === null ? projected.length : offset + limit;
    return projected.slice(offset, end);
  }

Performance: O(left × right)

With 10,000 books and 1,000 authors, the nested loop performs 10 million comparisons. Real databases use an index (B-tree) or a hash table to take "find the matching rows" from O(n) down to O(log n) or O(1). That's why JOIN columns are worth indexing.

Tests: JOIN semantic edges

The tests cover INNER / LEFT, multiple JOINs, JOIN + aggregation, ambiguous columns, duplicate aliases, cross joins without ON, and self joins. Those are the combinations you run into in practice.

tests/join.test.jsJavaScript · 84 lines · step 9
import test from "node:test";
import assert from "node:assert/strict";
import { Database } from "../src/database.js";
import { SqlError } from "../src/errors.js";

function makeDb() {
  const db = new Database();
  db.exec("CREATE TABLE authors (id INTEGER PRIMARY KEY, name TEXT);");
  db.exec("CREATE TABLE books (id INTEGER PRIMARY KEY, title TEXT, author_id INTEGER, price REAL);");
  db.exec("INSERT INTO authors VALUES (1, 'Alice'), (2, 'Bob'), (3, 'Carol');");
  db.exec("INSERT INTO books VALUES (1, 'JS', 1, 100), (2, 'DB', 1, 200), (3, 'OS', 2, 150);");
  return db;
}

test("INNER JOIN: keep only rows present on both sides", () => {
  const db = makeDb();
  const result = db.query(`SELECT books.title, authors.name
    FROM books JOIN authors ON books.author_id = authors.id
    ORDER BY books.id`);
  assert.deepEqual(result.rows, [["JS", "Alice"], ["DB", "Alice"], ["OS", "Bob"]]);
  assert.deepEqual(result.columns.map((c) => c.name), ["title", "name"]);
});

test("LEFT JOIN: keep rows with no right match, filling columns with NULL", () => {
  const db = makeDb();
  const result = db.query(`SELECT authors.name, books.title
    FROM authors LEFT JOIN books ON books.author_id = authors.id
    ORDER BY authors.id, books.id`);
  assert.deepEqual(result.rows, [
    ["Alice", "JS"],
    ["Alice", "DB"],
    ["Bob", "OS"],
    ["Carol", null],
  ]);
});

test("JOIN + WHERE + aggregate + GROUP BY", () => {
  const db = makeDb();
  const result = db.query(`SELECT authors.name, COUNT(*) AS n, SUM(books.price) AS total
    FROM books JOIN authors ON books.author_id = authors.id
    GROUP BY authors.name
    ORDER BY total DESC`);
  assert.deepEqual(result.rows, [["Alice", 2, 300], ["Bob", 1, 150]]);
});

test("multiple JOINs", () => {
  const db = makeDb();
  db.exec("CREATE TABLE reviews (book_id INTEGER, stars INTEGER);");
  db.exec("INSERT INTO reviews VALUES (1, 5), (1, 4), (2, 3);");
  const result = db.query(`SELECT books.title, COUNT(*) AS reviews
    FROM books
    JOIN authors ON books.author_id = authors.id
    JOIN reviews ON reviews.book_id = books.id
    GROUP BY books.title
    ORDER BY reviews DESC, books.title`);
  assert.deepEqual(result.rows, [["JS", 2], ["DB", 1]]);
});

test("identically named columns in a JOIN must be qualified", () => {
  const db = makeDb();
  assert.throws(() => db.query("SELECT id FROM books JOIN authors ON books.author_id = authors.id"), (err) => {
    assert.ok(err instanceof SqlError);
    assert.equal(err.code, "AMBIGUOUS_COLUMN");
    return true;
  });
});

test("duplicate table aliases are an error", () => {
  const db = makeDb();
  assert.throws(() => db.query("SELECT * FROM books JOIN books ON books.id = books.id"), /Duplicate table alias/);
});

test("a JOIN without ON is a cross join", () => {
  const db = makeDb();
  const result = db.query("SELECT COUNT(*) FROM authors JOIN books");
  assert.equal(result.rows[0][0], 9);
});

test("a self join requires an alias", () => {
  const db = makeDb();
  const result = db.query(`SELECT a.title AS first, b.title AS second
    FROM books a JOIN books b ON a.id < b.id`);
  assert.equal(result.rowCount, 3);
});

Chapter recap

  • JOIN is implemented with nested loops: for each left row, scan the right side.
  • LEFT JOIN keeps the left row when the right side has no match, filling right columns with NULL.
  • Columns with the same name need an alias; otherwise the engine raises AMBIGUOUS_COLUMN.
  • JOIN's performance bottleneck is O(n×m); indexes are the next step.

Exercises

  1. Implement RIGHT JOIN: swap the two sides, do a LEFT JOIN, then reorder the columns.
  2. Support JOIN ... USING (id): it means equality on the same-named columns, and the output keeps only one copy.
  3. Run a small experiment: with 5,000 rows, compare the time for a JOIN against narrowing the rows with WHERE first and then joining.

Chapter glossary

Nested loop join
An implementation that compares the two sides with two nested loops.
Driving table
The table scanned by the outer loop; usually the smaller one, or the one with fewer rows after filtering.
Cartesian product
A JOIN without an ON condition, producing left-rows × right-rows output.
